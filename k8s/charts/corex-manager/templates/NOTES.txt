   kubectl get svc {{ .Release.Name }}-corex -n {{ include "corex-manager.namespace" . }}

3. Access the frontend:
   {{- if eq .Values.corexService.type "LoadBalancer" }}

   {{- else if eq .Values.corexService.type "NodePort" }}

   curl http://localhost:8080/
   {{- else }}
